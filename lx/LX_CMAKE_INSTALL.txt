═══════════════════════════════════════════════════════════════════
  🚀 INSTALLATION (5 MINUTES)
═══════════════════════════════════════════════════════════════════

Step 1: Create Tools Directory
------------------------------
cd /Users/pedro/CLionProjects/os2
mkdir -p tools

Step 2: Copy Files
------------------
cp ~/Downloads/lxformat.h tools/
cp ~/Downloads/lxbuilder.c tools/
cp ~/Downloads/lxbuild.c tools/
cp ~/Downloads/makeexe.c tools/
cp ~/Downloads/makedll.c tools/
cp ~/Downloads/CMakeLists_tools.txt tools/CMakeLists.txt

Step 3: Build with CMake
------------------------
cd tools
mkdir build
cd build
cmake ..
make

Step 4: Test the Tools
---------------------
make test-lx

This creates:
  ✓ TEST.EXE  - Test executable
  ✓ TEST.DLL  - Test DLL

Step 5: Install (Optional)
--------------------------
cmake --install . --prefix ~/lxtools

This installs to ~/lxtools/bin:
  • lxbuild
  • makeexe
  • makedll

Add to PATH:
  echo 'export PATH="$HOME/lxtools/bin:$PATH"' >> ~/.bashrc
  source ~/.bashrc

Option 2: Integrated Build
--------------------------
Add tools to main project CMakeLists.txt:

Edit: /Users/pedro/CLionProjects/os2/CMakeLists.txt

Add at the end:

  option(BUILD_LX_TOOLS "Build LX executable tools" ON)
  if(BUILD_LX_TOOLS)
      add_subdirectory(tools)
  endif()

Then rebuild main project:
  cd /Users/pedro/CLionProjects/os2/build
  cmake ..
  make

Tools will be in: build/tools/

  💡 USAGE EXAMPLES
═══════════════════════════════════════════════════════════════════

After building, the tools are in the build directory:

Example 1: Create Simple Executable
-----------------------------------
cd /Users/pedro/CLionProjects/os2/tools/build

# Create test binary (ARM64 RET instruction)
printf '\xC0\x03\x5F\xD6' > hello.bin

# Create LX executable
./makeexe hello.bin HELLO.EXE 0

# Or use lxbuild for more control
./lxbuild -o HELLO.EXE --entry 0 --stack 64 hello.bin

Example 2: Create DLL with Exports
----------------------------------
# Create binary with functions at offsets 0, 16, 32
# (simplified - in reality you'd assemble ARM64 code)
dd if=/dev/zero of=mylib.bin bs=1 count=64

# Create DLL
./makedll mylib.bin MYLIB.DLL MYLIB Func1 Func2 Func3

# Or use lxbuild
./lxbuild -t dll -o MYLIB.DLL -n MYLIB \
    --export Func1:0 \
    --export Func2:16 \
    --export Func3:32 \
    mylib.bin

Example 3: Create with Imports
------------------------------
./lxbuild -o APP.EXE \
    --entry 0x1000 \
    --stack 128 \
    --import DOSCALLS:DosWrite \
    --import DOSCALLS:DosExit \
    app.bin

Example 4: Create Device Driver
-------------------------------
./lxbuild -t sys -o MYDRV.SYS driver.bin

═══════════════════════════════════════════════════════════════════
  🏗️  USING IN YOUR BUILD SYSTEM
═══════════════════════════════════════════════════════════════════

CMake Custom Command
--------------------
Add to your CMakeLists.txt:

# Build ARM64 binary and convert to LX executable
add_custom_command(
    OUTPUT myapp.exe
    COMMAND ${CMAKE_SOURCE_DIR}/tools/build/lxbuild
            -o myapp.exe
            --entry 0x1000
            --stack 64
            ${CMAKE_CURRENT_BINARY_DIR}/myapp.bin
    DEPENDS lxbuild myapp.bin
    COMMENT "Creating LX executable myapp.exe"
)

add_custom_target(myapp_lx ALL DEPENDS myapp.exe)

CMake Function for LX Executables
---------------------------------
Create function for easy LX creation:

function(add_lx_executable TARGET BINARY)
    add_custom_command(
        OUTPUT ${TARGET}.exe
        COMMAND lxbuild -o ${TARGET}.exe ${BINARY} ${ARGN}
        DEPENDS lxbuild ${BINARY}
        COMMENT "Creating LX ${TARGET}.exe"
    )
    add_custom_target(${TARGET}_lx ALL DEPENDS ${TARGET}.exe)
endfunction()

# Usage:
add_lx_executable(myapp myapp.bin --entry 0x1000)

═══════════════════════════════════════════════════════════════════
  🔨 DEVELOPMENT WORKFLOW
═══════════════════════════════════════════════════════════════════

Typical Workflow for Creating OS/2 Application:

1. Write ARM64 Assembly
   ----------------------
   # Create hello.s
   .global _start
   _start:
       mov x0, #0
       ret

2. Assemble to Binary
   -------------------
   aarch64-linux-gnu-as hello.s -o hello.o
   aarch64-linux-gnu-objcopy -O binary hello.o hello.bin

3. Create LX Executable
   --------------------
   cd /path/to/os2/tools/build
   ./lxbuild -o HELLO.EXE --entry 0 hello.bin

4. Test in QEMU
   ------------
   cd /path/to/os2/build
   make run
   # Boot OS/2 and run HELLO.EXE

═══════════════════════════════════════════════════════════════════
  🎯 PRACTICAL EXAMPLES
═══════════════════════════════════════════════════════════════════

Create System DLL (DOSCALLS.DLL)
--------------------------------
cd tools/build

# Compile system calls
aarch64-linux-gnu-as ../../syscalls.s -o syscalls.o
aarch64-linux-gnu-objcopy -O binary syscalls.o syscalls.bin

# Create DLL with all DOS APIs
./lxbuild -t dll -o DOSCALLS.DLL -n DOSCALLS \
    --export DosWrite:0 \
    --export DosRead:16 \
    --export DosOpen:32 \
    --export DosClose:48 \
    --export DosExit:64 \
    syscalls.bin

Create QuickBASIC Runtime
-------------------------
# Compile BASIC interpreter
./lxbuild -o BASIC.EXE \
    --entry 0x2000 \
    --stack 256 \
    --import DOSCALLS:DosWrite \
    --import DOSCALLS:DosExit \
    basic_runtime.bin

Create REXX Interpreter
-----------------------
./lxbuild -o REXX.EXE \
    --entry 0x1000 \
    --stack 128 \
    rexx_interpreter.bin

  🐛 TROUBLESHOOTING
═══════════════════════════════════════════════════════════════════

CMake Configuration Failed
  → Check CMake version: cmake --version (need 3.10+)
  → Delete build/ and try again
  → Check compiler: gcc --version

Build Errors
  → Make sure all source files are copied
  → Check file permissions
  → Try: make clean && make

Tests Fail
  → Check lxbuild is built: ls -la lxbuild
  → Run manually: ./lxbuild --help
  → Check test_code.bin exists

Can't Find Tools After Install
  → Check install prefix: cmake --install . --prefix ~/lxtools
  → Add to PATH: export PATH="$HOME/lxtools/bin:$PATH"
  → Verify: which lxbuild
