Step 4: Build tools
------------------
cd tools
make

Step 5: Create first executable
-------------------------------

═══════════════════════════════════════════════════════════════════
  📚 LX BUILDER API REFERENCE
═══════════════════════════════════════════════════════════════════

CREATING A BUILDER
------------------
LXBuilder* lx_builder_new(void);
  → Creates new LX builder instance

void lx_builder_free(LXBuilder *builder);
  → Frees builder and all resources

MODULE PROPERTIES
-----------------
void lx_builder_set_type(LXBuilder *builder, uint32_t flags);
  Flags:
    LXMF_PROGRAM         - Executable program (.EXE)
    LXMF_LIBRARY         - Dynamic library (.DLL)
    LXMF_PHYSICAL_DEVICE - Physical device driver (.SYS)
    LXMF_INIT_INSTANCE   - Per-process initialization
    LXMF_USES_PM         - Uses Presentation Manager

void lx_builder_set_name(LXBuilder *builder, const char *name);
  → Sets module name (for DLLs)

void lx_builder_set_version(LXBuilder *builder, uint32_t version);
  → Sets module version (e.g., 0x00010000 for 1.0)

void lx_builder_set_cpu(LXBuilder *builder, uint16_t cpu_type);
  Types:
    LXCPU_386    - Intel 386
    LXCPU_486    - Intel 486
    LXCPU_586    - Pentium
    LXCPU_ARM64  - ARM64 (for OS/2 Warp ARM64)

ADDING CODE AND DATA
--------------------
int lx_builder_add_object(LXBuilder *builder, const char *name,
                          const void *data, uint32_t size, uint32_t flags);
  
  Returns: Object index (0-based)
  
  Flags:
    LXOBJ_READABLE   - Read permission
    LXOBJ_WRITABLE   - Write permission
    LXOBJ_EXECUTABLE - Execute permission
    LXOBJ_SHARED     - Shared between processes
    LXOBJ_PRELOAD    - Preload on startup

  Example:
    // Add code section
    int code_obj = lx_builder_add_object(builder, ".text", code, code_size,
                                         LXOBJ_READABLE | LXOBJ_EXECUTABLE);
    
    // Add data section
    int data_obj = lx_builder_add_object(builder, ".data", data, data_size,
                                         LXOBJ_READABLE | LXOBJ_WRITABLE);

ENTRY POINT AND STACK
---------------------
void lx_builder_set_entry_point(LXBuilder *builder, uint32_t object,
                                uint32_t offset);
  → Sets the program entry point
  
  Example:
    lx_builder_set_entry_point(builder, 1, 0);  // Object 1, offset 0

void lx_builder_set_stack_size(LXBuilder *builder, uint32_t size);
  → Sets stack size in bytes
  
  Example:
    lx_builder_set_stack_size(builder, 0x10000);  // 64KB stack

EXPORTS (For DLLs)
------------------
int lx_builder_add_export(LXBuilder *builder, const char *name,
                          uint32_t object, uint32_t offset, uint16_t ordinal);
  
  Example:
    lx_builder_add_export(builder, "MyFunction", 1, 0x100, 1);

IMPORTS
-------
int lx_builder_add_import(LXBuilder *builder, const char *module,
                          const char *function, uint16_t ordinal);
  
  Example:
    lx_builder_add_import(builder, "DOSCALLS", "DosWrite", 0);

FIXUPS/RELOCATIONS
------------------
int lx_builder_add_fixup(LXBuilder *builder, uint32_t source_object,
                         uint32_t source_offset, uint32_t target_object,
                         uint32_t target_offset);

WRITING OUTPUT
--------------
int lx_builder_write(LXBuilder *builder, const char *filename);
  → Writes LX file to disk
  
  Returns: 0 on success, -1 on error

═══════════════════════════════════════════════════════════════════
  💡 EXAMPLE 1: CREATE SIMPLE EXECUTABLE
═══════════════════════════════════════════════════════════════════

#include "lxformat.h"

int main(void)
{
    // ARM64 code that calls DosExit(0)
    uint8_t code[] = {
        0x00, 0x00, 0x80, 0xD2,  // mov x0, #0
        0xC0, 0x03, 0x5F, 0xD6   // ret
    };
    
    // Create EXE
    lx_create_exe("MYAPP.EXE", code, sizeof(code), 0, 0x10000);
    
    return 0;
}

═══════════════════════════════════════════════════════════════════
  💡 EXAMPLE 2: CREATE DLL WITH EXPORTS
═══════════════════════════════════════════════════════════════════

#include "lxformat.h"

int main(void)
{
    // DLL code with functions at specific offsets
    uint8_t code[4096] = {
        // Function 1 at offset 0
        0xC0, 0x03, 0x5F, 0xD6,  // ret
        // Function 2 at offset 16
        // ...
    };
    
    // Export names
    const char *exports[] = {
        "MyFunction1",
        "MyFunction2",
        "MyFunction3"
    };
    
    // Create DLL
    lx_create_dll("MYLIB.DLL", "MYLIB", code, sizeof(code),
                  exports, 3);
    
    return 0;
}

═══════════════════════════════════════════════════════════════════
  💡 EXAMPLE 3: ADVANCED EXECUTABLE WITH MULTIPLE SECTIONS
═══════════════════════════════════════════════════════════════════

#include "lxformat.h"

int main(void)
{
    LXBuilder *builder = lx_builder_new();
    
    // Set properties
    lx_builder_set_type(builder, LXMF_PROGRAM);
    lx_builder_set_cpu(builder, LXCPU_ARM64);
    
    // Add code section
    uint8_t code[4096] = { /* ... */ };
    lx_builder_add_object(builder, ".text", code, sizeof(code),
                         LXOBJ_READABLE | LXOBJ_EXECUTABLE);
    
    // Add initialized data
    uint8_t data[2048] = { /* ... */ };
    lx_builder_add_object(builder, ".data", data, sizeof(data),
                         LXOBJ_READABLE | LXOBJ_WRITABLE);
    
    // Add BSS (uninitialized data)
    lx_builder_add_object(builder, ".bss", NULL, 8192,
                         LXOBJ_READABLE | LXOBJ_WRITABLE | LXOBJ_ZEROFILLED);
    
    // Set entry point
    lx_builder_set_entry_point(builder, 1, 0x100);
    
    // Set stack
    lx_builder_set_stack_size(builder, 0x20000);  // 128KB
    
    // Add imports
    lx_builder_add_import(builder, "DOSCALLS", "DosWrite", 0);
    lx_builder_add_import(builder, "DOSCALLS", "DosExit", 0);
    
    // Write output
    lx_builder_write(builder, "ADVANCED.EXE");
    
    lx_builder_free(builder);
    return 0;
}

═══════════════════════════════════════════════════════════════════
  💡 EXAMPLE 4: CREATE DEVICE DRIVER
═══════════════════════════════════════════════════════════════════

#include "lxformat.h"

int main(void)
{
    LXBuilder *builder = lx_builder_new();
    
    // Set as physical device driver
    lx_builder_set_type(builder, LXMF_PHYSICAL_DEVICE);
    lx_builder_set_name(builder, "MYDRIVER");
    
    // Add driver code
    uint8_t driver_code[8192] = { /* ... */ };
    lx_builder_add_object(builder, ".text", driver_code, sizeof(driver_code),
                         LXOBJ_READABLE | LXOBJ_EXECUTABLE | LXOBJ_RESIDENT);
    
    // Set entry point
    lx_builder_set_entry_point(builder, 1, 0);
    
    // Write .SYS file
    lx_builder_write(builder, "MYDRIVER.SYS");
    
    lx_builder_free(builder);
    return 0;
}

CMake Integration
-----------------
Add to your CMakeLists.txt:

# Build LX builder tool
add_executable(lxbuild
    tools/lxbuilder.c
    tools/lxformat.h
)

# Custom target to create LX executables
add_custom_command(
    OUTPUT ${CMAKE_BINARY_DIR}/myapp.exe
    COMMAND lxbuild --exe --entry 0x1000 --stack 0x10000
            -o ${CMAKE_BINARY_DIR}/myapp.exe
            ${CMAKE_BINARY_DIR}/myapp.o
    DEPENDS lxbuild myapp.o
    COMMENT "Creating LX executable myapp.exe"
)

═══════════════════════════════════════════════════════════════════

Offset   Size   Description
------   ----   -----------
0x0000   196    LX Header
         var    Object Table (sections)
         var    Object Page Table
         var    Resource Table (optional)
         var    Resident Names Table
         var    Entry Table
         var    Fixup Page Table
         var    Fixup Record Table
         var    Import Module Table
         var    Import Procedure Table
         var    Non-Resident Names Table
         var    Data Pages (actual code/data)
