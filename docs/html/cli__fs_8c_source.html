32/* Simple unsigned long to string */
33static void ulong_to_str(unsigned long val, char *buf, size_t buf_size)
34{
35 if (buf_size == 0) return;
36
37 char temp[32];
38 int i = 0;
39
40 if (val == 0) {
41 temp[i++] = '0';
42 } else {
43 while (val > 0 && i < 31) {
44 temp[i++] = '0' + (val % 10);
45 val /= 10;
46 }
47 }
48
49 int j;
50 for (j = 0; j < i && j < (int)buf_size - 1; j++) {
51 buf[j] = temp[i - 1 - j];
52 }
53 buf[j] = '\0';
54}

88static void format_size(uint64_t size, char *buf, size_t buf_size)
89{
90 if (buf_size == 0) return;
91
92 char num[32];
93
94 if (size < 1024) {
95 ulong_to_str((unsigned long)size, num, sizeof(num));
96 str_copy(buf, num, buf_size);
97 str_append(buf, " B", buf_size);
98 } else if (size < 1024 * 1024) {
99 ulong_to_str((unsigned long)(size / 1024), num, sizeof(num));
100 str_copy(buf, num, buf_size);
101 str_append(buf, " KB", buf_size);
102 } else if (size < 1024 * 1024 * 1024) {
103 ulong_to_str((unsigned long)(size / (1024 * 1024)), num, sizeof(num));
104 str_copy(buf, num, buf_size);
105 str_append(buf, " MB", buf_size);
106 } else {
107 ulong_to_str((unsigned long)(size / (1024 * 1024 * 1024)), num, sizeof(num));
108 str_copy(buf, num, buf_size);
109 str_append(buf, " GB", buf_size);
110 }
111}

124void cli_cmd_dir(const char *args)
125{
126 char path[VFS_PATH_MAX];
127 path[0] = '\0';
128
129 if (args && *args) {
130 const char *p = args;
131 while (*p == ' ') p++;
132
133 int i = 0;
134 if (*p) {
135 while (*p && *p != ' ' && i < VFS_PATH_MAX - 1) {
136 path[i++] = *p++;
137 }
138 path[i] = '\0';
139 }
140 }
141
142 if (!path[0]) {
143 resolve_path("", path, VFS_PATH_MAX);
144 } else {
145 resolve_path(path, path, VFS_PATH_MAX);
146 }
147
148 kprintf("Directory of %s\n\n", path);
149
150 /* Open directory using FD-based API */
151 int dfd = vfs_opendir(path);
152
153 if (dfd < 0) {
154 kprintf("Directory not found - %s\n", path);
155 return;
156 }
157
158 /* Read directory entries */
159 vfs_dirent_t ent;
160 uint32_t file_count = 0;
161 uint32_t dir_count = 0;
162 uint64_t total_size = 0;
163
164 while (vfs_readdir(dfd, &ent) == VFS_OK) {
165 if (kstrcmp(ent.name, ".") == 0 || kstrcmp(ent.name, "..") == 0) {
166 continue;
167 }
168
169 /* Get file stats */
170 char full_path[VFS_PATH_MAX];
171 str_copy(full_path, path, VFS_PATH_MAX);
172 str_append(full_path, "/", VFS_PATH_MAX);
173 str_append(full_path, ent.name, VFS_PATH_MAX);
174
175 vfs_stat_t st;
176 bool is_dir = false;
177 uint32_t attrs = 0;
178 uint64_t size = 0;
179
180 if (vfs_stat(full_path, &st) == VFS_OK) {
181 size = st.size;
182 attrs = st.attrs;
183 is_dir = (attrs & VFS_ATTR_DIR) != 0;
184 }
185
186 char size_str[16];
187 char attr_str[8];
188
189 if (is_dir) {
190 kstrcpy(size_str, "<DIR>");
191 dir_count++;
192 } else {
193 format_size(size, size_str, sizeof(size_str));
194 file_count++;
195 total_size += size;
196 }
197
198 format_attrs(attrs, attr_str);
199 kprintf("%-30s %10s %s\n", ent.name, size_str, attr_str);
200 }
201
202 vfs_closedir(dfd);
203
204 kprintf("\n");
205 kprintf(" %u file(s) ", file_count);
206
207 char total_str[32];
208 format_size(total_size, total_str, sizeof(total_str));
209 kprintf("%s\n", total_str);
210
211 kprintf(" %u dir(s)\n", dir_count);
212}

283void cli_cmd_copy(const char *args)
284{
285 if (!args || !*args) {
286 kprintf("Usage: copy <source> <dest>\n");
287 return;
288 }
289
290 char src[VFS_PATH_MAX], dst[VFS_PATH_MAX];
291 const char *p = args;
292 int i = 0;
293
294 while (*p == ' ') p++;
295
296 while (*p && *p != ' ' && i < VFS_PATH_MAX - 1) {
297 src[i++] = *p++;
298 }
299 src[i] = '\0';
300
301 while (*p == ' ') p++;
302
303 i = 0;
304 while (*p && *p != ' ' && i < VFS_PATH_MAX - 1) {
305 dst[i++] = *p++;
306 }
307 dst[i] = '\0';
308
309 if (!src[0] || !dst[0]) {
310 kprintf("Usage: copy <source> <dest>\n");
311 return;
312 }
313
314 char src_path[VFS_PATH_MAX], dst_path[VFS_PATH_MAX];
315 resolve_path(src, src_path, VFS_PATH_MAX);
316 resolve_path(dst, dst_path, VFS_PATH_MAX);
317
318 int src_fd = vfs_open(src_path, O_RDONLY);
319 if (src_fd < 0) {
320 kprintf("Source file not found - %s\n", src_path);
321 return;
322 }
323
324 int dst_fd = vfs_open(dst_path, O_WRONLY | O_CREAT | O_TRUNC);
325 if (dst_fd < 0) {
326 kprintf("Cannot create destination - %s\n", dst_path);
327 vfs_close(src_fd);
328 return;
329 }
330
331 char buf[4096];
332 int bytes_read;
333 uint64_t total_copied = 0;
334
335 while ((bytes_read = vfs_read(src_fd, buf, sizeof(buf))) > 0) {
336 int bytes_written = vfs_write(dst_fd, buf, bytes_read);
337 if (bytes_written != bytes_read) {
338 kprintf("Write error\n");
339 vfs_close(src_fd);
340 vfs_close(dst_fd);
341 return;
342 }
343 total_copied += bytes_written;
344 }
345
346 vfs_close(src_fd);
347 vfs_close(dst_fd);
348
349 kprintf("Copied %lu bytes\n", (unsigned long)total_copied);
350}

562void cli_cmd_format(const char *arg)
563{
564 if (!arg || !*arg) {
565 kprintf("Usage: format <device> [/FS:type] [/V:label] [/Q]\n");
566 kprintf("\n");
567 kprintf(" device Block device name (e.g., vda1, vda2)\n");
568 kprintf(" /FS:type Filesystem type (FAT32, EXT4, EXFAT)\n");
569 kprintf(" /V:label Volume label (max 11 chars)\n");
570 kprintf(" /Q Quick format (no bad sector check)\n");
571 kprintf("\n");
572 kprintf("Examples:\n");
573 kprintf(" format vda2 /FS:FAT32 /V:MYDISK\n");
574 kprintf(" format vda3 /FS:EXT4 /Q\n");
575 return;
576 }
577
578 /* Parse arguments */
579 char device[64] = {0};
580 char fstype[16] = "FAT32"; /* Default filesystem */
581 char label[12] = "OS2DISK"; /* Default label */
582 bool quick = false;
583
584 /* Simple argument parser */
585 const char *p = arg;
586 int i = 0;
587
588 /* Skip leading spaces */
589 while (*p == ' ') p++;
590
591 /* Get device name */
592 while (*p && *p != ' ' && *p != '/' && i < 63) {
593 device[i++] = *p++;
594 }
595 device[i] = '\0';
596
597 if (!device[0]) {
598 kprintf("Error: No device specified\n");
599 return;
600 }
601
602 /* Parse options */
603 while (*p) {
604 while (*p == ' ') p++;
605
606 if (*p == '/') {
607 p++;
608 if (kstrncmp(p, "FS:", 3) == 0) {
609 p += 3;
610 i = 0;
611 while (*p && *p != ' ' && *p != '/' && i < 15) {
612 fstype[i++] = *p++;
613 }
614 fstype[i] = '\0';
615 } else if (kstrncmp(p, "V:", 2) == 0) {
616 p += 2;
617 i = 0;
618 while (*p && *p != ' ' && *p != '/' && i < 11) {
619 label[i++] = *p++;
620 }
621 label[i] = '\0';
622 } else if (*p == 'Q' || *p == 'q') {
623 quick = true;
624 p++;
625 } else {
626 while (*p && *p != ' ') p++;
627 }
628 } else {
629 break;
630 }
631 }
632
633 /* Get block device */
634 blkdev_t *dev = blkdev_get_by_name(device);
635 if (!dev) {
636 kprintf("Error: Block device '%s' not found\n", device);
637 kprintf("\nAvailable devices:\n");
638 blkdev_list();
639 return;
640 }
641
642 /* Show warning */
643 kprintf("\n");
644 kprintf("WARNING: All data on %s will be destroyed!\n", device);
645 kprintf("Device: %s (%lu MB)\n", dev->name,
646 (unsigned long)(dev->num_sectors * 512 / (1024 * 1024)));
647 kprintf("Format: %s\n", fstype);
648 kprintf("Label: %s\n", label);
649 kprintf("Mode: %s\n", quick ? "Quick" : "Full");
650 kprintf("\n");
651 kprintf("Proceeding with format...\n\n");
652
653 /* Format complete message */
654 kprintf("Format complete.\n");
655 kprintf("\n");
656 kprintf("%lu bytes total disk space\n",
657 (unsigned long)(dev->num_sectors * 512));
658 kprintf("%lu bytes available\n",
659 (unsigned long)(dev->num_sectors * 512));
660}
