74 /* ─────────────────────────────────────────────────────────────────────

79 /* Get the block device for partition 2 (root) */

179 if (vfs_mount("/mnt/exfat", vda4, FS_EXFAT) == VFS_OK) {
180 uart_puts(" ✓ exFAT partition mounted (E:)\r\n");
181 }
182 }
183 } else {
184 uart_puts(" ✗ FAILED to mount root filesystem\r\n");
185 uart_puts(" Error: ");
186 uart_puts(vfs_errstr(mount_result));
187 uart_puts("\r\n\r\n");
188 uart_puts(" [VFS] Troubleshooting:\r\n");
189 uart_puts(" - Is EXT4 driver registered? (check vfs_init)\r\n");
190 uart_puts(" - Does block device have valid EXT4 filesystem?\r\n");
191 uart_puts(" - Run 'make mkdisk' to create disk image\r\n");
