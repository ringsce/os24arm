89static void print_uint(int64_t val, int base, bool upper, int pad, char padch)
90{
91 const char *digits = upper ? "0123456789ABCDEF" : "0123456789abcdef";
92 char buf[20];
93 int i = 0;
94
95 if (val == 0) {
96 buf[i++] = '0';
97 } else {
98 while (val) {
99 buf[i++] = digits[val % (int64_t)base];
100 val /= (int64_t)base;
101 }
102 }
103
104 /* Padding */
105 while (i < pad) buf[i++] = padch;
106
107 /* Reverse */
108 for (int j = 0; j < i / 2; j++) {
109 char tmp = buf[j]; buf[j] = buf[i-1-j]; buf[i-1-j] = tmp;
110 }
111 buf[i] = '\0';
112 uart_puts(buf);
113}

117void kprintf(const char *fmt, ...)
118{
119 va_list ap;
120 va_start(ap, fmt);
121
122 for (; *fmt; fmt++) {
123 if (*fmt != '%') { uart_putc(*fmt); continue; }
124 fmt++;
125
126 /* Width / padding */
127 char padch = ' ';
128 int width = 0;
129 if (*fmt == '0') { padch = '0'; fmt++; }
130 while (*fmt >= '0' && *fmt <= '9') { width = width*10 + (*fmt++ - '0'); }
131
132 /* Length modifier */
133 bool is_long = false;
134 if (*fmt == 'l') { is_long = true; fmt++; }
135
136 switch (*fmt) {
137 case 'c':
138 uart_putc((char)va_arg(ap, int));
139 break;
140 case 's': {
141 const char *s = va_arg(ap, const char *);
142 if (!s) s = "(null)";
143 uart_puts(s);
144 break;
145 }
146 case 'd': {
147 int64_t v = is_long ? va_arg(ap, int64_t) : va_arg(ap, int32_t);
148 if (v < 0) { uart_putc('-'); v = -v; }
149 print_uint((int64_t)v, 10, false, width, padch);
150 break;
151 }
152 case 'u': {
153 int64_t v = is_long ? va_arg(ap, int64_t) : va_arg(ap, uint32_t);
154 print_uint(v, 10, false, width, padch);
155 break;
156 }
157 case 'x': {
158 int64_t v = is_long ? va_arg(ap, int64_t) : va_arg(ap, uint32_t);
159 print_uint(v, 16, false, width, padch);
160 break;
161 }
162 case 'X': {
163 int64_t v = is_long ? va_arg(ap, int64_t) : va_arg(ap, uint32_t);
164 print_uint(v, 16, true, width, padch);
165 break;
166 }
167 case 'p': {
168 uintptr_t v = (uintptr_t)va_arg(ap, void *);
169 uart_puts("0x");
170 print_uint(v, 16, false, 16, '0');
171 break;
172 }
173 case '%':
174 uart_putc('%');
175 break;
176 default:
177 uart_putc('?');
178 break;
179 }
180 }
181 va_end(ap);
182}

186void kgets(char *buf, size_t len)
187{
188 size_t i = 0;
189 while (i < len - 1) {
190 char c = uart_getc();
191 if (c == '\r' || c == '\n') {
192 uart_puts("\r\n");
193 break;
194 }
195 if ((c == '\b' || c == 0x7f) && i > 0) {
196 /* Backspace */
197 uart_puts("\b \b");
198 i--;
199 continue;
200 }
201 if (c >= 0x20) {
202 uart_putc(c);
203 buf[i++] = c;
204 }
205 }
206 buf[i] = '\0';
207}
