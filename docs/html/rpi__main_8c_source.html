51static void uart_init(void)
52{
53 /* Disable UART */
54 *UART0_CR = 0;
55
56 /* Setup GPIO pins 14 and 15 for UART */
57 ULONG ra = *GPFSEL1;
58 ra &= ~((7 << 12) | (7 << 15)); /* Clear bits for GPIO 14 and 15 */
59 ra |= (4 << 12) | (4 << 15); /* Alt function 0 for UART */
60 *GPFSEL1 = ra;
61
62 /* Disable pull up/down for pins 14 and 15 */
63 *GPPUD = 0;
64 delay(150);
65 *GPPUDCLK0 = (1 << 14) | (1 << 15);
66 delay(150);
67 *GPPUDCLK0 = 0;
68
69 /* Clear pending interrupts */
70 *UART0_ICR = 0x7FF;
71
72 /* Set baud rate to 115200 */
73 /* UART clock = 48MHz, divisor = 48000000 / (16 * 115200) = 26.041666... */
74 /* Integer part = 26, fractional part = 0.041666... * 64 = 2.666... ≈ 3 */
75 *UART0_IBRD = 26;
76 *UART0_FBRD = 3;
77
78 /* Enable FIFO, 8-bit data, no parity, 1 stop bit */
79 *UART0_LCRH = (1 << 4) | (3 << 5);
80
81 /* Mask all interrupts */
82 *UART0_IMSC = 0;
83
84 /* Enable UART: transmit, receive, and UART enable */
85 *UART0_CR = (1 << 0) | (1 << 8) | (1 << 9);
86}
