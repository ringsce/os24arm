66static void hist_push(const char *line)
67{
68 if (!line[0]) return;
69 int prev = (hist_head + CLI_HIST_SIZE - 1) % CLI_HIST_SIZE;
70 if (hist_count > 0) {
71 int i = 0;
72 while (hist[prev][i] && hist[prev][i] == line[i]) i++;
73 if (!hist[prev][i] && !line[i]) return;

75 int j = 0;
76 while (line[j] && j < CLI_MAX_LINE - 1) {
77 hist[hist_head][j] = line[j];

93static int cli_readline(char *buf, int maxlen)
94{
95 int len = 0, cur = 0, hoff = 0;
96 char saved[CLI_MAX_LINE];
97 saved[0] = '\0';
98
99 for (;;) {
100 unsigned long key = kbd_getc();
101
102 if (key == KEY_CTRL_C) { puts_out("^C\r\n"); buf[0] = '\0'; return 0; }
103 if (key == KEY_CTRL_D && len == 0) { puts_out("exit\r\n"); return -1; }
104 if (key == KEY_CTRL_L) {
105 puts_out("\x1b[2J\x1b[H");
106 puts_out(cli_fs_get_prompt());
107 buf[len] = '\0'; puts_out(buf);
108 if (cur < len) cursor_left((unsigned long)(len - cur));
109 continue;
110 }
111
112 if (key == KEY_ENTER || key == '\n') {
113 buf[len] = '\0'; puts_out("\r\n"); hist_push(buf); return len;
114 }
115
116 if (key == KEY_BACKSPACE) {
117 if (cur == 0) continue;
118 kmemcpy(buf + cur - 1, buf + cur, len - cur);
119 len--; cur--; buf[len] = '\0';
120 cursor_left(1);
121 for (int i = cur; i < len; i++) putc_out(buf[i]);
122 erase_to_eol(); cursor_left((unsigned long)(len - cur));
123 continue;
124 }
125
126 if (key == KEY_DEL) {
127 if (cur == len) continue;
128 kmemcpy(buf + cur, buf + cur + 1, len - cur - 1);
129 len--; buf[len] = '\0';
130 for (int i = cur; i < len; i++) putc_out(buf[i]);
131 erase_to_eol(); cursor_left((unsigned long)(len - cur));
132 continue;
133 }
134
135 if (key == KEY_LEFT) { if (cur > 0) { cur--; cursor_left(1); } continue; }
136 if (key == KEY_RIGHT) { if (cur < len) { cur++; cursor_right(1); } continue; }
137 if (key == KEY_HOME) { cursor_left((unsigned long)cur); cur = 0; continue; }
138 if (key == KEY_END) { cursor_right((unsigned long)(len - cur)); cur = len; continue; }
139
140 if (key == KEY_UP) {
141 if (hoff == 0) { buf[len] = '\0'; kmemcpy(saved, buf, len + 1); }
142 const char *h = hist_get(hoff + 1);
143 if (!h) continue;
144 hoff++;
145 cursor_left((unsigned long)cur); erase_to_eol();
146 len = kstrlen(h); cur = len;
147 kmemcpy(buf, h, len + 1); puts_out(buf);
148 continue;
149 }
150
151 if (key == KEY_DOWN) {
152 if (hoff == 0) continue;
153 hoff--;
154 cursor_left((unsigned long)cur); erase_to_eol();
155 if (hoff == 0) {
156 len = kstrlen(saved); kmemcpy(buf, saved, len + 1);
157 } else {
158 const char *h = hist_get(hoff);
159 len = kstrlen(h); kmemcpy(buf, h, len + 1);
160 }
161 cur = len; puts_out(buf);
162 continue;
163 }
164
165 if (key == KEY_TAB) continue;
166 if (key > 0x7F) continue;
167
168 if (len >= maxlen - 1) continue;
169 kmemcpy(buf + cur + 1, buf + cur, len - cur);
170 buf[cur] = (char)key;
171 len++; cur++; buf[len] = '\0';
172 for (int i = cur - 1; i < len; i++) putc_out(buf[i]);
173 if (cur < len) cursor_left((unsigned long)(len - cur));
174 }
175}
