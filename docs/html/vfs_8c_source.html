25/* Find the best-matching mount for a path.
26 "C:/FOO" matches mount "C:", "/" matches mount "/".
27 Returns NULL if none. */
28static vfs_mount_t *find_mount(const char *path)
29{
30 vfs_mount_t *best = NULL;
31 int best_len = -1;
32
33 for (int i = 0; i < mount_count; i++) {
34 const char *mp = mounts[i].mountpoint;
35 int mpl = kstrlen(mp);
36
37 /* Match the mountpoint prefix */
38 if (kstrncmp(path, mp, (size_t)mpl) != 0) continue;
39
40 /* Next char must be '/', ':', or end of string */
41 char next = path[mpl];
42 if (next != '\0' && next != '/' && next != ':' && next != '\\')
43 continue;
44
45 if (mpl > best_len) { best = &mounts[i]; best_len = mpl; }
46 }
47 return best;
48}

50/* Strip the mountpoint prefix from a path, returning the relative path.
51 "C:/FOO/BAR" with mount "C:" → "/FOO/BAR" */
52static const char *relative_path(vfs_mount_t *mnt, const char *path)
53{
54 int mpl = kstrlen(mnt->mountpoint);
55 path += mpl;
56 /* skip a trailing ':' used in DOS-style paths */
57 if (*path == ':') path++;
58 if (*path == '\0') return "/";
59 return path;
60}

95int vfs_mount(const char *mountpoint, vfs_fs_t *fs, void *fs_priv)
96{
97 if (mount_count >= VFS_MAX_MOUNTS) return VFS_ERR_NOSPACE;
98
99 vfs_mount_t *mnt = &mounts[mount_count];
100 kstrncpy(mnt->mountpoint, mountpoint, VFS_PATH_MAX);
101 mnt->fs = fs;
102 mnt->priv = fs_priv;
103
104 int r = fs->mount(mnt);
105 if (r != VFS_OK) {
106 kmemset(mnt, 0, sizeof(*mnt));
107 kprintf("[VFS] Mount '%s' failed: %d\n", mountpoint, r);
108 return r;
109 }
110
111 mount_count++;
112 kprintf("[VFS] Mounted '%s' as '%s'\n", fs->name, mountpoint);
113 return VFS_OK;
114}

116int vfs_unmount(const char *mountpoint)
117{
118 for (int i = 0; i < mount_count; i++) {
119 if (kstrcmp(mounts[i].mountpoint, mountpoint) == 0) {
120 mounts[i].fs->unmount(&mounts[i]);
121 /* Shift table down */
122 for (int j = i; j < mount_count - 1; j++)
123 mounts[j] = mounts[j + 1];
124 mount_count--;
125 return VFS_OK;
126 }
127 }
128 return VFS_ERR_NOMOUNT;
129}

133int vfs_open(const char *path, int flags)
134{
135 vfs_mount_t *mnt = find_mount(path);
136 if (!mnt) return VFS_ERR_NOMOUNT;
137
138 int fd = alloc_fd();
139 if (fd < 0) return fd;
140
141 void *fpriv = NULL;
142 int r = mnt->fs->open(mnt, relative_path(mnt, path), flags, &fpriv);
143 if (r != VFS_OK) return r;
144
145 files[fd].used = true;
146 files[fd].mnt = mnt;
147 files[fd].fpriv = fpriv;
148 files[fd].flags = flags;
149 files[fd].pos = 0;
150 return fd;
151}

223{

226
227 int di = alloc_dfd();

230 void *dpriv = NULL;

234 dirs[di].used = true;
