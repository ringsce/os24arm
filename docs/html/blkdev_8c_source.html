59static int ramdisk_read(blkdev_t *dev, int64_t lba, uint32_t count, void *buf)
60{
61 ramdisk_priv_t *p = (ramdisk_priv_t *)dev->driver_data;
62 if (!p || !p->base) return -1;
63 if (lba + count > p->num_sectors) return -1;
64
65 kmemcpy(buf, p->base + lba * SECTOR_SIZE, count * SECTOR_SIZE);
66 return 0;
67}

79void ramdisk_create(const char *name, void *base, size_t size)
80{
81 ramdisk_priv.base = (uint8_t *)base;
82 ramdisk_priv.num_sectors = size / SECTOR_SIZE;
83
84 kstrcpy(ramdisk_dev.name, name);
85 ramdisk_dev.num_sectors = ramdisk_priv.num_sectors;
86 ramdisk_dev.sector_size = SECTOR_SIZE;
87 ramdisk_dev.driver_data = &ramdisk_priv;
88 ramdisk_dev.read = ramdisk_read;
89 ramdisk_dev.write = ramdisk_write;
90 ramdisk_dev.flags = 0;
91 ramdisk_dev.parent = NULL;
92 ramdisk_dev.start_sector = 0;
93
94 blkdev_register(&ramdisk_dev);
95 kprintf("[RAMDISK] Created: %s (%u KB)\n", name, (unsigned)(size / 1024));
96}
