39void *kmalloc(size_t size)
40{
41 if (size == 0) return NULL;
42 size = ALIGN8(size);
43
44 /* ── Scan free list ─────────────────────────────────────────────────── */
45 block_hdr_t **pp = &free_list;
46 while (*pp) {
47 block_hdr_t *b = *pp;
48 if (b->size >= size) {
49 /* Remove from free list */
50 *pp = b->next;
51 b->free = false;
52 b->next = NULL;
53 return (void *)((uintptr_t)b + HDR_SIZE);
54 }
55 pp = &b->next;
56 }
57
58 /* ── Bump allocate ──────────────────────────────────────────────────── */
59 if (heap_brk + HDR_SIZE + size > heap_limit) {
60 kprintf("[MEM] OUT OF MEMORY!\n");
61 return NULL;
62 }
63 block_hdr_t *b = (block_hdr_t *)heap_brk;
64 heap_brk += HDR_SIZE + size;
65
66 b->size = size;
67 b->free = false;
68 b->next = NULL;
69 return (void *)((uintptr_t)b + HDR_SIZE);
70}

72void kfree(void *ptr)
73{
74 if (!ptr) return;
75 block_hdr_t *b = (block_hdr_t *)((uintptr_t)ptr - HDR_SIZE);
76 if (b->free) {
77 kprintf("[MEM] double-free at %p!\n", ptr);
78 return;
79 }
80 b->free = true;
81 b->next = free_list;
82 free_list = b;
83}
