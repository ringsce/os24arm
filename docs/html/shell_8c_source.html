61/* ── Argument parsing ────────────────────────────────────────────────────── */
62static int parse_args(char *line, char *argv[])
63{
64 int argc = 0;
65 while (*line) {
66 while (*line == ' ') line++;
67 if (!*line) break;
68 argv[argc++] = line;
69 while (*line && *line != ' ') line++;
70 if (*line) *line++ = '\0';
71 if (argc >= MAX_ARGS) break;
72 }
73 return argc;
74}

151static void cmd_cd(int argc, char *argv[])
152{
153 if (argc < 2) {
154 kprintf("%s\n", cwd);
155 return;
156 }
157 /* Simulate navigation */
158 if (kstrcmp(argv[1], "..") == 0) {
159 /* Go up */
160 size_t len = kstrlen(cwd);
161 if (len > 3) { /* not root "C:\" */
162 cwd[len-1] = '\0'; /* remove trailing \ */
163 char *last = cwd;
164 while (kstrchr(last+1, '\\')) last = kstrchr(last+1, '\\');
165 *(last+1) = '\0';
166 }
167 } else if (kstrcmp(argv[1], "\\") == 0) {
168 kstrcpy(cwd, "C:\\");
169 } else {
170 /* Append subdirectory */
171 size_t clen = kstrlen(cwd);
172 size_t alen = kstrlen(argv[1]);
173 if (clen + alen + 2 < sizeof(cwd)) {
174 cwd_append(argv[1]);
175 cwd_append("\\");
176 }
177 }
178}

223static void cmd_shutdown(void)
224{
225 kprintf(ANSI_RED
226 "\n System halting. It is now safe to power off.\n\n"
227 ANSI_RESET);
228 /* PSCI SYSTEM_OFF (0x84000008) via HVC — works on QEMU virt */
229 __asm__ volatile(
230 "mov x0, #0x84000000\n"
231 "add x0, x0, #0x08\n"
232 "hvc #0\n"
233 );
234 /* If PSCI is not available, spin forever */
235 while (1) {
236 __asm__ volatile("wfe");
237 }

278
279void shell_run(void)
280{
281 cmd_cls();
282 cmd_ver();
283
284 kprintf("Type " ANSI_CYAN "HELP" ANSI_RESET " for available commands.\n\n");
285
286 static char line[MAX_CMD_LEN];
287 static char *argv[MAX_ARGS];
288
289 for (;;) {
290 draw_prompt();
291 kgets(line, sizeof(line));
292
293 if (line[0] == '\0') continue;
294 hist_push(line);
295
296 /* Uppercase the command token for case-insensitive matching */
297 for (size_t i = 0; line[i] && line[i] != ' '; i++) {
298 if (line[i] >= 'a' && line[i] <= 'z') line[i] -= 32;
299 }
300
301 int argc = parse_args(line, argv);
302 if (argc == 0) continue;
303
304 const char *cmd = argv[0];
305
306 if (kstrcmp(cmd, "HELP") == 0) cmd_help();
307 else if (kstrcmp(cmd, "CLS") == 0) cmd_cls();
308 else if (kstrcmp(cmd, "VER") == 0) cmd_ver();
309 else if (kstrcmp(cmd, "DIR") == 0) cmd_dir();
310 else if (kstrcmp(cmd, "CD") == 0) cmd_cd(argc, argv);
311 else if (kstrcmp(cmd, "ECHO") == 0) cmd_echo(argc, argv);
312 else if (kstrcmp(cmd, "SET") == 0) cmd_set();
313 else if (kstrcmp(cmd, "UNAME")== 0) cmd_uname();
314 else if (kstrcmp(cmd, "MEM") == 0) mem_dump();
315 else if (kstrcmp(cmd, "DATE") == 0) cmd_date();
316 else if (kstrcmp(cmd, "HISTORY")==0) cmd_history();
317 else if (kstrcmp(cmd, "TYPE") == 0) cmd_type(argc, argv);
318 else if (kstrcmp(cmd, "MD") == 0) cmd_md(argc, argv);
319 else if (kstrcmp(cmd, "BASIC")== 0) basic_run();
320 else if (kstrcmp(cmd, "SHUTDOWN")==0) cmd_shutdown();
321 else {
322 kprintf("'%s' is not recognized as an internal command.\n"
323 "Type HELP for a list of commands.\n", cmd);
324 }
325 }
326
327 if (ends_with_cmd(input_buffer)) {
328
329 static char script_buffer[4096];
330
331 int size = fs_read_file(input_buffer,
332 script_buffer,
333 sizeof(script_buffer));
334
335 if (size < 0) {
336 kputs("File not found\n");
337 } else {
338 rexx_execute(script_buffer);
339 }
340
341 } else {
342 rexx_execute(input_buffer);
343 }
344
345}
